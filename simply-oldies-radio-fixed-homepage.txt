<script>
(function () {
  "use strict";

  /* =========================================================
     0. MOVE SOURCE WIDGETS INTO HERO & HIDE LEFTOVER CONTAINERS
     ========================================================= */

  function moveAndHideSourceButtons() {
    const hub = document.getElementById("sor-home-hub");
    const liveMount = document.getElementById("sor-hero-live-mount");
    const scheduleMount = document.getElementById("sor-hero-schedule-mount");
    const liveFallback = document.getElementById("sor-hero-live-fallback");
    const scheduleFallback = document.getElementById("sor-hero-schedule-fallback");

    if (!hub) return;

    const allSources = Array.from(
      document.querySelectorAll(".so-live-btn-source, .so-schedule-btn-source")
    );

    let liveSource = null;
    let scheduleSource = null;

    allSources.forEach((widget) => {
      const isScheduleClass = widget.classList.contains("so-schedule-btn-source");
      const text = (widget.textContent || "").toLowerCase();
      const link = widget.querySelector("a");
      const href = link ? (link.getAttribute("href") || "").toLowerCase() : "";

      if (isScheduleClass || href.includes("schedule") || text.includes("schedule")) {
        if (!scheduleSource) scheduleSource = widget;
        else widget.style.setProperty("display", "none", "important");
      } else if (!liveSource) {

  function stripHtml(html) {
    const tmp = document.createElement("div");

  function timeAgo(dateStr) {
    const ts = Date.parse(dateStr);

    if (isNaN(ts)) return "";

    const diffMins = Math.max(
      1,
      Math.floor((Date.now() - ts) / 60000)
    );

    if (diffMins < 60) return diffMins + "m ago";

    const diffHrs = Math.floor(diffMins / 60);

    if (diffHrs < 24) return diffHrs + "h ago";

    return Math.floor(diffHrs / 24) + "d ago";
  }

  function classifyArticle(title, description) {
    const haystack = (title + " " + description).toLowerCase();

    if (INAPPROPRIATE_NEWS_TERMS.some((badTerm) => haystack.includes(badTerm))) {
      return { matched: false, era: "all", label: "" };
    }

    for (const rule of ERA_RULES) {
      if (rule.terms.some((term) => haystack.includes(term))) {
        return {
          matched: true,
          era: rule.era,
          label: rule.label
        };
      }
    }

    return { matched: false, era: "all", label: "" };
  }

  async function fetchLiveNews(forceRefresh) {
    if (!newsListEl) return;

    if (!forceRefresh) {
      try {
        const cachedRaw = localStorage.getItem(NEWS_CACHE_KEY);

        if (cachedRaw) {
          const cached = JSON.parse(cachedRaw);

          if (
            cached &&
            Date.now() - cached.timestamp < NEWS_CACHE_TTL &&
            Array.isArray(cached.articles) &&
            cached.articles.length
          ) {
            allFilteredArticles = cached.articles
              .map((article) => {
                const check = classifyArticle(
                  article.title || "",
                  article.excerpt || ""
                );

                if (!check.matched) return null;

                return {
                  ...article,
                  era: check.era,
                  eraLabel: check.label
                };
              })
              .filter(Boolean);

            try {
              localStorage.setItem(
                NEWS_CACHE_KEY,
                JSON.stringify({
                  timestamp: Date.now(),
                  articles: allFilteredArticles
                })
              );
            } catch (error) {}

            if (allFilteredArticles.length) {
              renderNews();

              if (newsUpdatedEl) {
                newsUpdatedEl.textContent =
                  "Updated " + timeAgo(new Date(cached.timestamp).toISOString());
              }

              return;
            }
          }
        }
      } catch (error) {}

        const endpoint =
          "https://api.rss2json.com/v1/api.json?rss_url=" +
          encodeURIComponent(src.url);

        const res = await fetch(endpoint);

        if (!res.ok) throw new Error("Feed error");

        const data = await res.json();

        if (data.status !== "ok" || !Array.isArray(data.items)) return [];

        return data.items.map((entry) => {
          const cleanTitle = stripHtml(entry.title);
          const cleanExcerpt = stripHtml(entry.description || entry.content);

          const check = classifyArticle(cleanTitle, cleanExcerpt);

          if (!check.matched) return null;

          let thumb = entry.thumbnail || "";

          if (!thumb && entry.enclosure && entry.enclosure.link) {
            thumb = entry.enclosure.link;
          }

          return {
            title: cleanTitle,
            excerpt: cleanExcerpt,
            link: entry.link,
            pubDate: entry.pubDate,
            source: src.name,
            thumbnail: thumb,
            era: check.era,
            eraLabel: check.label
          };
        }).filter(Boolean);
      })
    );

    const merged = [];
    const seenTitles = new Set();

    results.forEach((result) => {
      if (result.status === "fulfilled" && Array.isArray(result.value)) {

          if (!seenTitles.has(key)) {
            seenTitles.add(key);
            merged.push(article);
          }
        });
      }
    });

    merged.sort(
      (a, b) => (Date.parse(b.pubDate) || 0) - (Date.parse(a.pubDate) || 0)
    );

      allFilteredArticles = merged;

      try {
        localStorage.setItem(
          NEWS_CACHE_KEY,
          JSON.stringify({
            timestamp: Date.now(),
            articles: merged
          })
        );
      } catch (error) {}

      renderNews();

      if (newsUpdatedEl) newsUpdatedEl.textContent = "Updated just now";

        '<div class="sor-news-loading">Unable to load live feeds right now. Click Refresh to try again.</div>';
    }

let currentArtworkUrl = "";

function showFallbackArtwork() {

    nowPlayingArtFallbackEl.style.display = "block";

function showTrackArtwork(url) {
  if (!nowPlayingArtEl || !url) return;

  nowPlayingArtEl.hidden = false;

  if (nowPlayingArtFallbackEl) {
    nowPlayingArtFallbackEl.style.display = "none";
  }
}

function syncProRadioArtwork() {
  if (!nowPlayingArtEl) return;

  const playerArtwork = document.querySelector(
    "#qtmplayer .qtmplayer__cover img"
  );

  if (!playerArtwork) return;

  const artworkUrl =
    playerArtwork.currentSrc ||
    playerArtwork.getAttribute("src") ||
    playerArtwork.getAttribute("data-src") ||
    "";

  if (!artworkUrl) {
    showFallbackArtwork();
    return;
  }

  if (artworkUrl === currentArtworkUrl) {
    if (
      nowPlayingArtEl.complete &&
      nowPlayingArtEl.naturalWidth > 0
    ) {
      showTrackArtwork(artworkUrl);
    }
    return;
  }

  currentArtworkUrl = artworkUrl;

  nowPlayingArtEl.onload = () => {
    if (nowPlayingArtEl.src === artworkUrl) {
      showTrackArtwork(artworkUrl);
    }
  };

  nowPlayingArtEl.onerror = () => {
    if (currentArtworkUrl === artworkUrl) {
      showFallbackArtwork();
    }
  };

  nowPlayingArtEl.src = artworkUrl;

  if (
    nowPlayingArtEl.complete &&
    nowPlayingArtEl.naturalWidth > 0
  ) {
    showTrackArtwork(artworkUrl);
  }
}

/* Observe Pro.Radio player creation and artwork changes */

let proRadioPlayerObserver = null;
let proRadioArtworkObserver = null;

function attachProRadioArtworkObserver() {
  const player = document.getElementById("qtmplayer");

  if (!player || proRadioArtworkObserver) return;

  proRadioArtworkObserver = new MutationObserver(() => {
    syncProRadioArtwork();
  });

  proRadioArtworkObserver.observe(player, {
    subtree: true,
    childList: true,
    attributes: true,
    attributeFilter: ["src", "data-src"]
  });

  syncProRadioArtwork();
}

function findProRadioPlayer() {
  if (document.getElementById("qtmplayer")) {
    attachProRadioArtworkObserver();

    if (proRadioPlayerObserver) {
      proRadioPlayerObserver.disconnect();
      proRadioPlayerObserver = null;
    }
  }
}

findProRadioPlayer();

  proRadioPlayerObserver = new MutationObserver(findProRadioPlayer);

  proRadioPlayerObserver.observe(document.documentElement, {
    childList: true,
    subtree: true
  });
}

  }
}
