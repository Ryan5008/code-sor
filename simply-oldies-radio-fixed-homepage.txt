function syncProRadioArtwork() {
  if (!nowPlayingArtEl) return;

  const playerArtwork = document.querySelector(
    "#qtmplayer .qtmplayer__cover img"
  );

  if (!playerArtwork) return;

  const artworkUrl =
    playerArtwork.currentSrc ||
    playerArtwork.getAttribute("src") ||
    playerArtwork.getAttribute("data-src") ||
    "";

  if (!artworkUrl) {
    showFallbackArtwork();
    return;
  }

  if (artworkUrl === currentArtworkUrl) {
    if (
      nowPlayingArtEl.complete &&
      nowPlayingArtEl.naturalWidth > 0
    ) {
      showTrackArtwork(artworkUrl);
    }
    return;
  }

  currentArtworkUrl = artworkUrl;

  nowPlayingArtEl.onload = () => {
    if (nowPlayingArtEl.src === artworkUrl) {
      showTrackArtwork(artworkUrl);
    }
  };

  nowPlayingArtEl.onerror = () => {
    if (currentArtworkUrl === artworkUrl) {
      showFallbackArtwork();
    }
  };

  nowPlayingArtEl.src = artworkUrl;

  if (
    nowPlayingArtEl.complete &&
    nowPlayingArtEl.naturalWidth > 0
  ) {
    showTrackArtwork(artworkUrl);
  }
}

/* Observe Pro.Radio player creation and artwork changes */

let proRadioPlayerObserver = null;
let proRadioArtworkObserver = null;

function attachProRadioArtworkObserver() {
  const player = document.getElementById("qtmplayer");

  if (!player || proRadioArtworkObserver) return;

  proRadioArtworkObserver = new MutationObserver(() => {
    syncProRadioArtwork();
  });

  proRadioArtworkObserver.observe(player, {
    subtree: true,
    childList: true,
    attributes: true,
    attributeFilter: ["src", "data-src"]
  });

  syncProRadioArtwork();
}

function findProRadioPlayer() {
  if (document.getElementById("qtmplayer")) {
    attachProRadioArtworkObserver();

    if (proRadioPlayerObserver) {
      proRadioPlayerObserver.disconnect();
      proRadioPlayerObserver = null;
    }
  }
}

        `https://api.open-meteo.com/v1/forecast?latitude=${numLat.toFixed(4)}` +
        `&longitude=${numLon.toFixed(4)}` +
        `&current_weather=true` +
        `&hourly=relative_humidity_2m,apparent_temperature` +
        `&daily=weather_code,temperature_2m_max,temperature_2m_min` +
        `&wind_speed_unit=mph&timezone=auto`;

      const res = await fetch(url);

      if (!res.ok) {
        throw new Error("Open-Meteo HTTP " + res.status);
      }

      const data = await res.json();

      if (!data || !data.current_weather) {
        throw new Error("Missing current_weather");
      }

      const cw = data.current_weather;

      let humidity = 72;
      let feelsLike = cw.temperature;

      if (data.hourly && Array.isArray(data.hourly.time)) {
        const hourPrefix = String(cw.time || "").slice(0, 13);

        const idx = data.hourly.time.findIndex((time) =>
          time.startsWith(hourPrefix)
        );

        const useIdx = idx >= 0 ? idx : 0;

        if (
          Array.isArray(data.hourly.relative_humidity_2m) &&
          data.hourly.relative_humidity_2m[useIdx] != null
        ) {
          humidity = data.hourly.relative_humidity_2m[useIdx];
        }

        if (
          Array.isArray(data.hourly.apparent_temperature) &&
          data.hourly.apparent_temperature[useIdx] != null
        ) {
          feelsLike = data.hourly.apparent_temperature[useIdx];
        }
      }

      const dailyDays = [];

      if (data.daily && Array.isArray(data.daily.time)) {
        data.daily.time.forEach((date, i) => {
          dailyDays.push({
            date: date,
            code: data.daily.weather_code
              ? data.daily.weather_code[i]
              : 2,
            hi: data.daily.temperature_2m_max
              ? data.daily.temperature_2m_max[i]
              : cw.temperature,
            lo: data.daily.temperature_2m_min
              ? data.daily.temperature_2m_min[i]
              : cw.temperature
          });
        });
      }

      renderWeatherUI(
        label,
        cw.temperature,
        feelsLike,
        cw.windspeed,
        humidity,
        cw.weathercode,
        dailyDays
      );

      return true;

    } catch (err) {
      /* Secondary weather provider fallback */
      try {
        const wttrUrl =
          `https://wttr.in/${numLat.toFixed(3)},${numLon.toFixed(3)}?format=j1`;

        const wttrRes = await fetch(wttrUrl);

        if (!wttrRes.ok) {
          throw new Error("wttr fallback failed");
        }

        const wData = await wttrRes.json();
        const cur = wData.current_condition && wData.current_condition[0];

        if (!cur) throw new Error("No wttr current");

        const dailyDays = (wData.weather || []).map((day) => ({
          date: day.date,
          code: 2,
          hi: Number(day.maxtempC),
          lo: Number(day.mintempC)
        }));

        renderWeatherUI(
          label,
          Number(cur.temp_C),
          Number(cur.FeelsLikeC),
          Number(cur.windspeedMiles),
          Number(cur.humidity),
          2,
          dailyDays
        );

        return true;

      } catch (fallbackErr) {
        setIdleWeatherPrompt(
          label,
          "Click Locate Me to see your local weather"
        );

        return false;
      }
    }

  function getDecadeLabel(year) {
    const y = Number(year);

    if (y >= 1950 && y <= 1959) return "1950s Milestone";
    if (y >= 1960 && y <= 1969) return "1960s Milestone";
    if (y >= 1970 && y <= 1979) return "1970s Milestone";
    if (y >= 1980 && y <= 1989) return "1980s Milestone";
    if (y >= 1990 && y <= 1999) return "1990s Milestone";

    return "50s–90s Milestone";
  }

  function scoreMusicEntry(ev, isBirthList) {
    const y = Number(ev.year);

    if (y < 1950 || y > 1999) return -1;

    const lower = String(ev.text || "").toLowerCase();

    if (NON_MUSIC_BLOCKLIST.some((term) => lower.includes(term))) {
      return -1;
    }

    const hasMusicWord = MUSIC_CONTEXT_WORDS.some((word) =>
      lower.includes(word)
    );

    const hasIconicArtist = ICONIC_50S_90S_ARTISTS.some((artist) =>
      lower.includes(artist)
    );

    if (!hasMusicWord && !hasIconicArtist) return -1;

    if (isBirthList) return hasIconicArtist ? 2 : -1;

    if (hasMusicWord && hasIconicArtist) return 5;
    if (hasMusicWord) return 4;

    return 3;
  }

  function escapeMusicText(value) {
    return String(value || "")
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/&lt;(\/?(?:strong|em))&gt;/gi, "<$1>");
  }

  function renderOnThisDay(year, text, imageUrl) {
    if (!otdBodyEl) return;

    const safeImage =
      imageUrl && /^https?:\/\//i.test(imageUrl)
        ? `<img class="sor-otd-image" src="${imageUrl}" alt="Photograph relating to this day in music history" loading="lazy" onerror="this.remove()">`
        : "";

    otdBodyEl.innerHTML = `
      <div class="sor-otd-top-row">
        <span class="sor-otd-year-badge">📅 ${year}</span>
        <span class="sor-otd-era-label">${getDecadeLabel(year)}</span>
      </div>
      <p class="sor-otd-text">${escapeMusicText(text)}</p>
      ${safeImage}
    `;
  }

  /*
   * Retrieve a thumbnail from Wikipedia when the On This Day
   * event itself does not include an image.
   */

  async function getWikipediaImage(pageTitle) {
    if (!pageTitle) return "";

    try {
      const endpoint =
        "https://en.wikipedia.org/api/rest_v1/page/summary/" +
        encodeURIComponent(pageTitle.replace(/ /g, "_"));

      const response = await fetch(endpoint);

      if (!response.ok) return "";

      const data = await response.json();

      if (data.thumbnail && data.thumbnail.source) {
        return data.thumbnail.source;
      }

      if (data.originalimage && data.originalimage.source) {
        return data.originalimage.source;
      }

      return "";
    } catch (error) {
      return "";
    }
  }

  async function loadOnThisDayInMusic() {
    const formatter = new Intl.DateTimeFormat("en-GB", {
      timeZone: "Europe/London",
      month: "2-digit",
      day: "2-digit"
    });

    const parts = {};

    formatter.formatToParts(new Date()).forEach((part) => {
      parts[part.type] = part.value;
    });

    const mm = parts.month || "11";
    const dd = parts.day || "20";

    const displayLabel = new Date(
      Date.UTC(2026, Number(mm) - 1, Number(dd), 12)
    ).toLocaleDateString("en-GB", {
      timeZone: "UTC",
      day: "numeric",
      month: "short"
    }).toUpperCase();

    if (otdDateBadge) {
      otdDateBadge.textContent = displayLabel;
    }

    /* New cache version prevents old image-less entries persisting */
    const cacheKey = `sor_otd_music_v6_${mm}_${dd}`;

    try {
      const cached = localStorage.getItem(cacheKey);

      if (cached) {
        const parsed = JSON.parse(cached);

        if (
          parsed &&
          parsed.year &&
          parsed.text &&
          parsed.image
        ) {
          renderOnThisDay(parsed.year, parsed.text, parsed.image);
          return;
        }
      }
    } catch (error) {}

    try {
      const res = await fetch(
        `https://en.wikipedia.org/api/rest_v1/feed/onthisday/all/${mm}/${dd}`
      );

      if (!res.ok) {
        throw new Error("Wikipedia OTD HTTP " + res.status);
      }

      const data = await res.json();

      const eventPool = [
        ...(Array.isArray(data.selected) ? data.selected : []),
        ...(Array.isArray(data.events) ? data.events : [])
      ].map((event) => ({
        ev: event,
        score: scoreMusicEntry(event, false)
      })).filter((item) => item.score > 0);

      const birthPool = (
        Array.isArray(data.births) ? data.births : []
      ).map((event) => ({
        ev: event,
        score: scoreMusicEntry(event, true)
      })).filter((item) => item.score > 0);

      const ranked = (eventPool.length ? eventPool : birthPool)
        .sort((a, b) => b.score - a.score);

      if (ranked.length) {
        /*
         * Prefer events with an available image rather than choosing
         * an image-less event simply because it scored highly.
         */

        const imageCandidates = ranked.filter((item) => {
          const pages = Array.isArray(item.ev.pages)
            ? item.ev.pages
            : [];

          return pages.some((page) =>
            page.thumbnail && page.thumbnail.source
          );
        });

        const candidates = imageCandidates.length
          ? imageCandidates
          : ranked;

        const topScore = candidates[0].score;

        const topTier = candidates.filter(
          (item) => item.score === topScore
        );

        const chosen =
          topTier[Number(dd) % topTier.length].ev;

        const cleanText = String(chosen.text || "").trim();

        const pageWithImage = Array.isArray(chosen.pages)
          ? chosen.pages.find((page) =>
              page.thumbnail && page.thumbnail.source
            )
          : null;

        let imageUrl = pageWithImage
          ? pageWithImage.thumbnail.source
          : "";

        /*
         * If the chosen event has no thumbnail, try its Wikipedia page.
         */

        if (!imageUrl && Array.isArray(chosen.pages) && chosen.pages.length) {
          const pageTitle = chosen.pages[0].title;

          imageUrl = await getWikipediaImage(pageTitle);
        }

        const result = {
          year: chosen.year,
          text: cleanText,
          image: imageUrl
        };

        /*
         * Only cache results when an image was found.
         * This allows another attempt if Wikipedia has no image today.
         */

        if (imageUrl) {
          try {
            localStorage.setItem(
              cacheKey,
              JSON.stringify(result)
            );
          } catch (error) {}
        }

        renderOnThisDay(
          chosen.year,
          cleanText,
          imageUrl
        );

        return;
      }

    } catch (error) {}

    /*
     * Special date fallback for 20 November.
     */

    if (mm === "11" && dd === "20") {
      const todayEvent =
        NOV_20_MUSIC_EVENTS[
          new Date().getDate() % NOV_20_MUSIC_EVENTS.length
        ];

      let imageUrl = await getWikipediaImage(todayEvent.page);

      renderOnThisDay(
        todayEvent.year,
        todayEvent.text,
        imageUrl
      );

      return;
    }

    /*
     * General fallback if Wikipedia has no suitable event.
     */

    const fallback =
      NOV_20_MUSIC_EVENTS[
        (Number(dd) + Number(mm)) % NOV_20_MUSIC_EVENTS.length
      ];

    const fallbackImage = await getWikipediaImage(fallback.page);

    renderOnThisDay(
      fallback.year,
      fallback.text,
      fallbackImage
    );
  }

  loadOnThisDayInMusic();
